\section{Rdhaounia, N., y Elweriemmi, M. (2026). The complex trilogy of shadow economy, inflation, and economic growth: Evidence from MENA countries. European Journal of Government and Economics, 15(1), 55-75. https://doi.org/10.17979/ejge.2026.15.1.12398}

\begin{frame}
\paperframe
    \begin{block}{01 \quad Objetivo}
        Analizar empíricamente las interrelaciones dinámicas y la causalidad estructural entre la economía informal, la inflación y el crecimiento económico en 15 países de la región MENA durante el periodo 1991--2017.
    \end{block}
    \begin{alertblock}{Pregunta de investigación}
       ¿Existe causalidad bidireccional entre la economía informal, la inflación y el crecimiento económico al controlar los problemas de endogeneidad y simultaneidad estructural en la región MENA?
    \end{alertblock}
\end{frame}

\end{column}
\begin{column}{0.42\textwidth}
\small
Mayor crecimiento \textbf{reduce} la informalidad, porque expande el empleo formal y los incentivos al cumplimiento.
\begin{itemize}
  \item Crecimiento $\rightarrow$ informalidad: negativo en ambos métodos.
  \item Informalidad $\rightarrow$ crecimiento: negativo \textbf{solo en GMM}; en 3SLS no es significativo.
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{column}{0.42\textwidth}
\small
Vínculo causal \textbf{bidireccional positivo}: ambas se refuerzan.
\begin{itemize}
  \item La informalidad erosiona los ingresos públicos e impulsa el financiamiento monetario, lo que eleva la inflación.
  \item La inflación fomenta la informalidad, porque los agentes buscan protegerse de la pérdida de poder adquisitivo.
\end{itemize}
\end{column}

\begin{column}{0.42\textwidth}
\small
Relación \textbf{bidireccional negativa} entre inflación y crecimiento.
\begin{itemize}
  \item Una inflación moderada puede estimular la actividad, pero una inflación alta y persistente frena el crecimiento por distorsión de precios y menores ingresos reales.
\end{itemize}
\end{column}

\begin{frame}
\paperframe
\begin{block}{05 \quad Limitaciones del artículo}
\small
\begin{itemize}
  \item \textbf{Condición de orden mal calculada:} usan $g'=1$, pero cada ecuación contiene las 3 endógenas ($g'=3$). La ecuación de inflación está \textbf{exactamente} identificada, no sobreidentificada.
  \item \textbf{Condición de rango:} se cumple en 3SLS (Self.Empl, MS, TR y FDI significativos), pero en GMM la IED no es significativa: la ecuación de inflación queda débilmente identificada.
  \item \textbf{Exclusiones discutibles:} la recaudación y la IED quedan fuera de la ecuación de informalidad; el autoempleo explica la informalidad casi de forma mecánica.
  \item \textbf{Otros:} $R^2$ negativo en la ecuación de inflación (3SLS), no reportan efectos fijos, no indican la fuente de la mayoría de variables y los datos terminan en 2017.
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{¿Cuáles son los siguientes pasos que debo seguir?}
\begin{enumerate}
  \item[\textbf{01}] Construir la tasa de empleo informal y la tasa de desempleo con la ENAHO (2004--2019), y validar la informalidad con la variable oficial \texttt{ocupinf} del INEI (desde 2012).
  \item[\textbf{02}] Empalmar el VAB departamental del INEI (base 1994 y base 2007), expresarlo en términos per cápita y obtener la inversión pública ejecutada por departamento (MEF).
  \item[\textbf{03}] Verificar la condición de rango: probar que el desempleo y la inversión pública son relevantes en su ecuación (F de primera etapa $>$ 10).
  \item[\textbf{04}] Estimar 2SLS y 3SLS con efectos fijos, y aplicar las pruebas de Hausman, endogeneidad y sobreidentificación.
\end{enumerate}
